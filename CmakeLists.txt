cmake_minimum_required(VERSION 3.14)
project(HomeFlix VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Include headers from "include"
include_directories(${PROJECT_SOURCE_DIR}/include)

# Fetch Catch2
include(FetchContent)
FetchContent_Declare(
  catch2
  GIT_REPOSITORY https://github.com/catchorg/Catch2.git
  GIT_TAG v3.4.0
)
FetchContent_MakeAvailable(catch2)

# Main executable source files (excluding test files)
file(GLOB_RECURSE SOURCES "src/*.cpp")
list(APPEND SOURCES "${PROJECT_SOURCE_DIR}/main.cpp")
add_executable(main ${SOURCES})
target_link_libraries(main PRIVATE ws2_32)
target_link_options(main PRIVATE -mconsole)

# Test executable
file(GLOB TEST_SOURCES "test/*.cpp")
add_executable(MyTests ${TEST_SOURCES})
target_link_libraries(MyTests
  PRIVATE
    Catch2::Catch2WithMain
    ws2_32
)
target_include_directories(MyTests PRIVATE ${PROJECT_SOURCE_DIR}/include)

# Enable CTest and Catch2 integration
enable_testing()
include(CTest)
include(Catch)
catch_discover_tests(MyTests)
